"use client";

import { useState } from "react";
import { createClient } from "@/lib/supabase/client";
import { Input } from "@/components/ui/input";
import { Save, MessageSquarePlus, CheckCircle, Loader2 } from "lucide-react";

export default function GerenciarChamadoForm({ chamado, prestadores }: { chamado: any, prestadores: any[] }) {
  const [status, setStatus] = useState(chamado.status || "aberto");
  const [sla, setSla] = useState(chamado.sla || "");
  const [prestadorId, setPrestadorId] = useState(chamado.prestador_id ? String(chamado.prestador_id) : "");
  const [novoComentario, setNovoComentario] = useState("");
  const [loading, setLoading] = useState(false);
  const [sucesso, setSucesso] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    setErro(null);
    setSucesso(false);

    const supabase = createClient();

    // 1. Atualiza o chamado
    const { error: updateError } = await supabase.from("chamados").update({
      status,
      sla: sla || null,
      prestador_id: prestadorId ? Number(prestadorId) : null,
    }).eq("id", chamado.id);

    if (updateError) {
      setErro("Erro ao atualizar: " + updateError.message);
      setLoading(false);
      return;
    }

    // 2. Salva o histórico
    if (novoComentario.trim() !== "") {
      const { data: { user } } = await supabase.auth.getUser();
      
      const { error: histError } = await supabase.from("chamados_historico").insert([{
        chamado_id: chamado.id,
        autor_id: user?.id,
        texto: novoComentario
      }]);

      if (histError) {
        setErro("Erro ao salvar histórico: " + histError.message);
        setLoading(false);
        return;
      }
    }

    // 3. DISPARO DE E-MAIL (AQUI ESTÁ O PULO DO GATO)
    try {
      console.log("Iniciando disparo de e-mail...");
      
      const { data: autor } = await supabase.from("profiles").select("email").eq("id", chamado.autor_id).single();
      const { data: sindicos } = await supabase.from("profiles").select("email").eq("role", "sindico");
      
      const destinatarios = [];
      if (autor?.email) destinatarios.push(autor.email);
      sindicos?.forEach(s => { if (s.email) destinatarios.push(s.email); });

      if (destinatarios.length > 0 && novoComentario.trim() !== "") {
        const response = await fetch('/api/enviar-email', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            to: destinatarios,
            subject: `Atualização no Chamado: ${chamado.titulo}`,
            html: `<h2>Atualização de Chamado</h2>
                   <p><strong>Chamado:</strong> ${chamado.titulo}</p>
                   <p><strong>Nova Anotação:</strong> ${novoComentario}</p>
                   <p>Status atual: ${status}</p>`
          })
        });
        console.log("Resposta da API de e-mail:", response.status);
      }
    } catch (err) {
      console.error("Falha ao disparar e-mail:", err);
    }

    setLoading(false);
    setSucesso(true);
    
    setTimeout(() => {
      window.location.href = "/chamados";
    }, 1500);
  };

  return (
    <div className="flex flex-col gap-5">
      {erro && (
        <div className="p-3 bg-red-100 text-red-700 text-sm rounded-md border border-red-200">
          {erro}
        </div>
      )}

      <form onSubmit={handleSubmit} className="flex flex-col gap-5">
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          <div>
            <label className="text-sm font-medium mb-1 block text-slate-300">Status</label>
            <select className="flex h-10 w-full rounded-md border border-slate-600 bg-slate-700 text-white px-3 py-2 text-sm" value={status} onChange={(e) => setStatus(e.target.value)}>
              <option value="aberto">Aberto</option>
              <option value="em_andamento">Em Andamento</option>
              <option value="resolvido">Resolvido</option>
            </select>
          </div>

          <div>
            <label className="text-sm font-medium mb-1 block text-slate-300">SLA (Data Limite)</label>
            <input 
              type="date" 
              className="flex h-10 w-full rounded-md border border-slate-600 bg-slate-700 text-white px-3 py-2 text-sm"
              value={sla} 
              onChange={(e) => setSla(e.target.value)} 
            />
          </div>
        </div>

        <div>
          <label className="text-sm font-medium mb-1 block text-slate-300">Prestador Responsável</label>
          <select className="flex h-10 w-full rounded-md border border-slate-600 bg-slate-700 text-white px-3 py-2 text-sm" value={prestadorId} onChange={(e) => setPrestadorId(e.target.value)}>
            <option value="">Nenhum</option>
            {prestadores.map((p) => (
              <option key={p.id} value={p.id}>{p.nome} - {p.especialidade}</option>
            ))}
          </select>
        </div>

        <div className="bg-slate-900/50 p-4 rounded-lg border border-slate-700">
          <label className="text-sm font-bold mb-2 flex items-center gap-2 text-slate-200">
            <MessageSquarePlus className="w-4 h-4" /> Nova Anotação (Opcional)
          </label>
          <textarea 
            className="flex min-h-[80px] w-full rounded-md border border-slate-600 bg-slate-700 text-white px-3 py-2 text-sm placeholder:text-slate-400" 
            placeholder="Ex: Prestador agendou para amanhã às 14h..."
            value={novoComentario}
            onChange={(e) => setNovoComentario(e.target.value)}
          />
        </div>

        <button 
          type="submit" 
          disabled={loading || sucesso} 
          className={`w-full h-11 rounded-md flex items-center justify-center font-bold text-base transition-colors ${
            sucesso ? "bg-green-600 text-white" : "bg-blue-600 hover:bg-blue-500 text-white"
          }`}
        >
          {loading ? (
            <><Loader2 className="w-5 h-5 mr-2 animate-spin" /> Salvando...</>
          ) : sucesso ? (
            <><CheckCircle className="w-5 h-5 mr-2" /> Salvo com Sucesso!</>
          ) : (
            <><Save className="w-5 h-5 mr-2" /> Salvar Alterações</>
          )}
        </button>
      </form>
    </div>
  );
}

import { createClient } from "@/lib/supabase/server";
import { redirect } from "next/navigation";
import Link from "next/link";
import { Button } from "@/components/ui/button";
import { ArrowLeft, MapPin, Clock, MessageSquare, User, Tag, AlertCircle, Calendar, Wrench } from "lucide-react";
import GerenciarChamadoForm from "./form";
import { format } from "date-fns";
import { ptBR } from "date-fns/locale";

export async function GerenciarChamado({ params }: { params: Promise<{ id: string }> }) {
  const { id } = await params;

  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  
  if (!user) redirect("/auth/login");

  const { data: profile } = await supabase.from("profiles").select("role").eq("id", user.id).single();
  if (profile?.role !== "sindico") redirect("/chamados");

  const { data: chamado, error } = await supabase
    .from("chamados")
    .select(`
      *,
      autor:profiles(nome, unidade),
      tipos_servico(nome),
      sub_servicos(nome),
      prestadores(nome)
    `)
    .eq("id", id)
    .single();

  const { data: prestadores } = await supabase.from("prestadores").select("*").order("nome");

  const { data: historico } = await supabase
    .from("chamados_historico")
    .select(`*, profiles (nome)`)
    .eq("chamado_id", id)
    .order("created_at", { ascending: true });

  if (error || !chamado) {
    return (
      <div className="text-center p-12 bg-slate-800 rounded-xl border border-slate-700 shadow-sm">
        <h2 className="text-red-400 font-bold text-xl mb-2">Chamado não encontrado</h2>
        <p className="text-slate-400 text-sm mb-6">O ID {id} não retornou nenhum resultado.</p>
        <Link href="/chamados">
          <Button variant="outline" className="text-slate-200 border-slate-600 hover:bg-slate-700">Voltar para a lista</Button>
        </Link>
      </div>
    );
  }

  return (
    <div className="w-full max-w-2xl flex flex-col gap-6 pb-24">
      
      {/* CABEÇALHO */}
      <div className="flex items-center gap-2 border-b border-slate-700 pb-4">
        <Link href="/chamados">
          <Button variant="ghost" size="icon" className="-ml-2 text-slate-400 hover:text-white"><ArrowLeft className="w-5 h-5" /></Button>
        </Link>
        <h1 className="text-xl font-bold text-white">Gerenciar Chamado #{chamado.id}</h1>
      </div>

      {/* DETALHES DO CHAMADO - FUNDO ESCURO */}
      <div className="bg-slate-800 p-6 rounded-xl shadow-xl text-white border border-slate-700">
        <div className="flex justify-between items-start mb-3">
          <h2 className="font-bold text-2xl text-white">{chamado.titulo}</h2>
          <span className={`text-xs px-3 py-1 rounded-full font-bold ${
            chamado.status === 'aberto' ? 'bg-orange-500 text-white' :
            chamado.status === 'em_andamento' ? 'bg-blue-500 text-white' :
            'bg-green-500 text-white'
          }`}>
            {chamado.status.replace('_', ' ').toUpperCase()}
          </span>
        </div>
        
        <div className="flex flex-wrap gap-3 text-sm mb-4 pb-4 border-b border-slate-600">
          <span className="flex items-center gap-1.5 bg-slate-700 px-3 py-1.5 rounded-full text-slate-200">
            <User className="w-4 h-4 text-amber-400" /> 
            Aberto por: <strong className="text-white">{chamado.autor?.nome || "Morador"}</strong> ({chamado.autor?.unidade})
          </span>
          <span className="flex items-center gap-1.5 bg-slate-700 px-3 py-1.5 rounded-full text-slate-200">
            <Tag className="w-4 h-4 text-amber-400" /> 
            {chamado.tipos_servico?.nome} {chamado.sub_servicos?.nome && `- ${chamado.sub_servicos.nome}`}
          </span>
          <span className="flex items-center gap-1.5 bg-slate-700 px-3 py-1.5 rounded-full text-slate-200">
            <Clock className="w-4 h-4 text-amber-400" /> 
            {format(new Date(chamado.created_at), "dd/MM/yyyy 'às' HH:mm", { locale: ptBR })}
          </span>
        </div>

        <p className="text-slate-200 whitespace-pre-wrap leading-relaxed mb-4">{chamado.descricao}</p>
        
        <div className="flex flex-wrap gap-3 mt-4 pt-4 border-t border-slate-600">
          {chamado.localizacao && (
            <div className="text-sm text-amber-200 flex items-center gap-2 bg-slate-900/50 px-3 py-1.5 rounded-lg border border-slate-700">
              <MapPin className="w-4 h-4 text-amber-400" /> <strong>Local:</strong> {chamado.localizacao}
            </div>
          )}
          {chamado.sla && (
            <div className="text-sm text-amber-200 flex items-center gap-2 bg-slate-900/50 px-3 py-1.5 rounded-lg border border-slate-700">
              <Calendar className="w-4 h-4 text-amber-400" /> <strong>SLA:</strong> {format(new Date(chamado.sla), "dd/MM/yyyy")}
            </div>
          )}
          {chamado.prestadores && (
            <div className="text-sm text-amber-200 flex items-center gap-2 bg-slate-900/50 px-3 py-1.5 rounded-lg border border-slate-700">
              <Wrench className="w-4 h-4 text-amber-400" /> <strong>Prestador:</strong> {chamado.prestadores.nome}
            </div>
          )}
        </div>
      </div>

      {/* HISTÓRICO DE MOVIMENTAÇÕES */}
      <div className="bg-slate-700 p-6 rounded-xl border border-slate-600 shadow-lg">
        <h3 className="font-bold text-lg text-white flex items-center gap-2 mb-4 border-b border-slate-600 pb-2">
          <MessageSquare className="w-5 h-5 text-blue-400" /> Histórico de Movimentações
        </h3>
        
        {!historico || historico.length === 0 ? (
          <p className="text-slate-300 text-sm italic flex items-center gap-2">
            <AlertCircle className="w-4 h-4" /> Nenhuma movimentação registrada ainda.
          </p>
        ) : (
          <div className="flex flex-col gap-4">
            {historico.map((h: any) => (
              <div key={h.id} className="flex gap-3">
                <div className="flex flex-col items-center">
                  <div className="w-8 h-8 bg-blue-500/20 text-blue-300 rounded-full flex items-center justify-center shrink-0 border border-blue-500/30">
                    <User className="w-4 h-4" />
                  </div>
                  <div className="w-px h-full bg-slate-600 my-1"></div>
                </div>
                <div className="pb-4 w-full">
                  <div className="flex items-center gap-2 mb-1">
                    <span className="font-bold text-sm text-white">{h.profiles?.nome || "Sistema"}</span>
                    <span className="text-xs text-slate-400">
                      {format(new Date(h.created_at), "dd/MM/yyyy 'às' HH:mm", { locale: ptBR })}
                    </span>
                  </div>
                  <div className="bg-slate-800 border border-slate-600 p-3 rounded-md text-sm text-slate-200 whitespace-pre-wrap shadow-sm">
                    {h.texto}
                  </div>
                </div>
              </div>
            ))}
          </div>
        )}
      </div>

      {/* FORMULÁRIO / BOTÃO FLUTUANTE */}
      <GerenciarChamadoForm chamado={chamado} prestadores={prestadores || []} />
    </div>
  );
}

"use client";

import { useState, useEffect } from "react";
import { createClient } from "@/lib/supabase/client";
import Link from "next/link";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { ArrowLeft, Send, MapPin } from "lucide-react";

export default function NovoChamadoPage() {
  const [titulo, setTitulo] = useState("");
  const [descricao, setDescricao] = useState("");
  const [localizacao, setLocalizacao] = useState("");
  const [tipoId, setTipoId] = useState("");
  const [subId, setSubId] = useState("");
  const [tipos, setTipos] = useState<any[]>([]);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    async function carregar() {
      const supabase = createClient();
      const { data } = await supabase.from("tipos_servico").select("*, sub_servicos(*, prestador_id)").order("nome");
      if (data && data.length > 0) {
        setTipos(data);
        setTipoId(data[0].id.toString());
      }
    }
    carregar();
  }, []);

  const tipoSelecionado = tipos.find(t => t.id.toString() === tipoId);
  const subServicos = tipoSelecionado?.sub_servicos || [];
  const subSelecionado = subServicos.find((s: any) => s.id.toString() === subId);

  const handlePegarLocalizacao = () => {
    if (!navigator.geolocation) {
      alert("Seu navegador não suporta geolocalização.");
      return;
    }
    navigator.geolocation.getCurrentPosition(
      (position) => {
        const { latitude, longitude } = position.coords;
        setLocalizacao(`https://www.google.com/maps?q=${latitude},${longitude}`);
      },
      (error) => {
        alert("Não foi possível obter a localização. Verifique se o GPS está ligado.");
      }
    );
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    const supabase = createClient();
    const { data: { user } } = await supabase.auth.getUser();

    if (!user) {
      alert("Usuário não autenticado.");
      setLoading(false);
      return;
    }

    const { error } = await supabase.from("chamados").insert([{
      autor_id: user.id,
      titulo,
      descricao,
      localizacao,
      tipo_servico_id: Number(tipoId),
      sub_servico_id: subId ? Number(subId) : null,
      prestador_id: subSelecionado?.prestador_id || null,
      status: subSelecionado?.prestador_id ? "em_andamento" : "aberto"
    }]);

    setLoading(false);
    
    if (error) {
      alert("Erro ao criar chamado: " + error.message);
    } else {
      window.location.href = "/chamados";
    }
  };

  return (
    <div className="flex-1 w-full flex flex-col items-center p-4 md:p-12 bg-slate-900 min-h-screen">
      <div className="w-full max-w-2xl bg-slate-800 p-6 md:p-8 rounded-xl border border-slate-700 shadow-lg">
        
        <div className="flex items-center gap-2 mb-6 border-b border-slate-700 pb-4">
          <Link href="/chamados">
            <Button variant="ghost" size="icon" className="-ml-2 text-slate-400 hover:text-white"><ArrowLeft className="w-5 h-5" /></Button>
          </Link>
          <h1 className="text-xl font-bold text-white">Abrir Chamado</h1>
        </div>

        <form onSubmit={handleSubmit} className="flex flex-col gap-4">
          <div>
            <label className="text-sm font-medium mb-1 block text-slate-300">Tipo de Serviço</label>
            <select 
              className="flex h-10 w-full rounded-md border border-slate-600 bg-slate-700 text-white px-3 py-2 text-sm" 
              value={tipoId} 
              onChange={(e) => { setTipoId(e.target.value); setSubId(""); }} 
              required
            >
              {tipos.length === 0 && <option value="">Carregando...</option>}
              {tipos.map(t => <option key={t.id} value={t.id}>{t.nome}</option>)}
            </select>
          </div>

          {subServicos.length > 0 && (
            <div>
              <label className="text-sm font-medium mb-1 block text-slate-300">O que exatamente está acontecendo?</label>
              <select 
                className="flex h-10 w-full rounded-md border border-slate-600 bg-slate-700 text-white px-3 py-2 text-sm" 
                value={subId} 
                onChange={(e) => setSubId(e.target.value)}
              >
                <option value="">Selecione (Opcional)</option>
                {subServicos.map((s: any) => <option key={s.id} value={s.id}>{s.nome}</option>)}
              </select>
            </div>
          )}

          <div>
            <label className="text-sm font-medium mb-1 block text-slate-300">Título do Problema</label>
            <Input required placeholder="Ex: Vazamento na pia da cozinha" value={titulo} onChange={(e) => setTitulo(e.target.value)} className="bg-slate-700 border-slate-600 text-white" />
          </div>
          
          <div>
            <label className="text-sm font-medium mb-1 block text-slate-300">Descrição detalhada (Opcional)</label>
            <textarea className="flex min-h-[100px] w-full rounded-md border border-slate-600 bg-slate-700 text-white px-3 py-2 text-sm" value={descricao} onChange={(e) => setDescricao(e.target.value)} />
          </div>
          
          <div>
            <label className="text-sm font-medium mb-1 block text-slate-300">Localização</label>
            <div className="flex gap-2">
              <Input placeholder="Descreva o local ou cole o link do Maps" value={localizacao} onChange={(e) => setLocalizacao(e.target.value)} className="bg-slate-700 border-slate-600 text-white" />
              <button type="button" onClick={handlePegarLocalizacao} className="bg-slate-700 hover:bg-slate-600 text-slate-300 px-3 rounded-md border border-slate-600 flex items-center justify-center" title="Pegar minha localização atual">
                <MapPin className="w-5 h-5" />
              </button>
            </div>
          </div>

          <Button type="submit" disabled={loading} className="mt-4 bg-orange-600 hover:bg-orange-500 text-white h-11 text-base">
            {loading ? "Registrando..." : <><Send className="w-4 h-4 mr-2" /> Registrar Chamado</>}
          </Button>
        </form>
      </div>
    </div>
  );
}

        <div className="flex items-center gap-2 mb-6 border-b border-slate-700 pb-4">

            <Button variant="ghost" size="icon" className="-ml-2 text-slate-400 hover:text-white"><ArrowLeft className="w-5 h-5" /></Button>
          </Link>
          <h1 className="text-xl font-bold text-white">Abrir Chamado</h1>

        <form onSubmit={handleSubmit} className="flex flex-col gap-4">

            <select 
              className="flex h-10 w-full rounded-md border border-slate-600 bg-slate-700 text-white px-3 py-2 text-sm" 
              value={tipoId} 
              onChange={(e) => { setTipoId(e.target.value); setSubId(""); }} 
              required
            >
              {tipos.length === 0 && <option value="">Carregando...</option>}
              {tipos.map(t => <option key={t.id} value={t.id}>{t.nome}</option>)}
            </select>
          </div>

          {subServicos.length > 0 && (
            <div>
              <label className="text-sm font-medium mb-1 block text-slate-300">O que exatamente está acontecendo?</label>
              <select 
                className="flex h-10 w-full rounded-md border border-slate-600 bg-slate-700 text-white px-3 py-2 text-sm" 
                value={subId} 
                onChange={(e) => setSubId(e.target.value)}
              >
                <option value="">Selecione (Opcional)</option>
                {subServicos.map((s: any) => <option key={s.id} value={s.id}>{s.nome}</option>)}
              </select>
            </div>
          )}

          <div>
            <label className="text-sm font-medium mb-1 block text-slate-300">Título do Problema</label>
            <Input required placeholder="Ex: Vazamento na pia da cozinha" value={titulo} onChange={(e) => setTitulo(e.target.value)} className="bg-slate-700 border-slate-600 text-white" />
          </div>
          
          <div>
            <label className="text-sm font-medium mb-1 block text-slate-300">Descrição detalhada (Opcional)</label>
            <textarea className="flex min-h-[100px] w-full rounded-md border border-slate-600 bg-slate-700 text-white px-3 py-2 text-sm" value={descricao} onChange={(e) => setDescricao(e.target.value)} />
          </div>
          
          <div>
            <label className="text-sm font-medium mb-1 block text-slate-300">Localização</label>
            <div className="flex gap-2">
              <Input placeholder="Descreva o local ou cole o link do Maps" value={localizacao} onChange={(e) => setLocalizacao(e.target.value)} className="bg-slate-700 border-slate-600 text-white" />
              <button type="button" onClick={handlePegarLocalizacao} className="bg-slate-700 hover:bg-slate-600 text-slate-300 px-3 rounded-md border border-slate-600 flex items-center justify-center" title="Pegar minha localização atual">
                <MapPin className="w-5 h-5" />
              </button>
            </div>
          </div>

          <Button type="submit" disabled={loading} className="mt-4 bg-orange-600 hover:bg-orange-500 text-white h-11 text-base">
            {loading ? "Registrando..." : <><Send className="w-4 h-4 mr-2" /> Registrar Chamado</>}
          </Button>
        </form>
      </div>
    </div>
  );
}

import { NextResponse } from 'next/server';

// O Resend pega a chave do .env.local automaticamente
const resend = new Resend(process.env.RESEND_API_KEY);

export async function POST(request: Request) {
  try {
    // Log para verificar se a chave está chegando
    console.log("CHAVE DO RESEND:", process.env.RESEND_API_KEY ? "Configurada" : "AUSENTE");

    const { subject, html } = await request.json();

    // DESTINATÁRIO DE TESTE (o seu e-mail)
    const to = "xiko57@gmail.com";
    
    // REMETENTE (usando o seu domínio verificado)
    const from = "Nova Suíça 02 <nao-responda@mail.sostoyota.com.br>";

    const { data, error } = await resend.emails.send({
      from: from,
      to: to,
      subject: subject,
      html: html,
    });

    if (error) {
      console.error("ERRO DO RESEND:", error);
      return NextResponse.json({ error: error.message }, { status: 400 });
    }

    return NextResponse.json({ sucesso: true, id: data?.id });

  } catch (error: any) {
    console.error("ERRO INTERNO:", error);
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}
